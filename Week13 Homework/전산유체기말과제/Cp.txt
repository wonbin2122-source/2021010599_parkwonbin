import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path
import numpy as np
import re

# ==================================================
# 1. 에어포일 chord 자동 탐지 (farfield 제거용)
# ==================================================
def find_airfoil_chord(df, x_col, y_col):
    y_abs = df[y_col].abs()
    y_lim = np.quantile(y_abs, 0.30)
    surf = df[y_abs <= y_lim]

    if len(surf) < 500:
        y_lim = np.quantile(y_abs, 0.45)
        surf = df[y_abs <= y_lim]

    x = surf[x_col].to_numpy()
    hist, edges = np.histogram(x, bins=300)
    th = hist.max() * 0.06
    idx = np.where(hist >= th)[0]

    if len(idx) > 0:
        x_le = edges[idx.min()]
        x_te = edges[idx.max() + 1]
    else:
        x_le = np.quantile(x, 0.05)
        x_te = np.quantile(x, 0.95)

    return float(x_le), float(x_te)


# ==================================================
# 2. LE 부분만 자연스럽게 연결 (기말과제용)
# ==================================================
def connect_le_for_report(df, Cp_col, x_col="x_c", limit_x=0.05):
    df = df.copy()
    mask = df[x_col] <= limit_x
    df.loc[mask, Cp_col] = (
        df.loc[mask, Cp_col]
        .interpolate(method="linear", limit=3)
    )
    return df


# ==================================================
# 3. 메인 코드
# ==================================================
BASE_DIR = Path(__file__).resolve().parent
files = sorted(BASE_DIR.glob("*deg.xlsx"))

print(f"찾은 Cp 파일 개수: {len(files)}")
for f in files:
    print(" -", f.name)

for file in files:
    aoa = re.findall(r"\d+", file.name)[0]
    df = pd.read_excel(file)

    # 컬럼명 (SU2 기준)
    x_col  = "Points_0"
    y_col  = "Points_1"
    Cp_col = "Pressure_Coefficient"

    # ------------------------------------------------
    # chord 정의
    # ------------------------------------------------
    x_le, x_te = find_airfoil_chord(df, x_col, y_col)
    c = x_te - x_le

    df["x_c"] = (df[x_col] - x_le) / c
    df = df[(df["x_c"] >= 0.0) & (df["x_c"] <= 1.0)].copy()

    # ------------------------------------------------
    # 상 / 하면 분리
    # ------------------------------------------------
    upper = df[df[y_col] >= 0].copy()
    lower = df[df[y_col] < 0].copy()

    # ------------------------------------------------
    # x/c binning + median
    # ------------------------------------------------
    nbins = 70
    bins = np.linspace(0.0, 1.0, nbins + 1)

    upper["bin"] = pd.cut(upper["x_c"], bins)
    lower["bin"] = pd.cut(lower["x_c"], bins)

    upper_b = upper.groupby("bin").median(numeric_only=True)
    lower_b = lower.groupby("bin").median(numeric_only=True)

    # bin 중심값
    upper_b["x_c"] = [iv.mid for iv in upper_b.index]
    lower_b["x_c"] = [iv.mid for iv in lower_b.index]

    upper_b = upper_b.dropna().sort_values("x_c")
    lower_b = lower_b.dropna().sort_values("x_c")

    # ------------------------------------------------
    # LE 부분만 부드럽게 연결 (기말과제용 핵심)
    # ------------------------------------------------
    upper_b = connect_le_for_report(upper_b, Cp_col)
    lower_b = connect_le_for_report(lower_b, Cp_col)

    # ------------------------------------------------
    # Plot
    # ------------------------------------------------
    plt.figure(figsize=(6, 4))

    plt.plot(
        upper_b["x_c"], upper_b[Cp_col],
        "o-", markersize=3, linewidth=1.3,
        label="Upper surface"
    )
    plt.plot(
        lower_b["x_c"], lower_b[Cp_col],
        "s-", markersize=3, linewidth=1.3,
        label="Lower surface"
    )

    plt.gca().invert_yaxis()
    plt.xlabel("x/c")
    plt.ylabel("Cp")
    plt.title(f"Cp Distribution (AoA = {aoa}°)")
    plt.legend()
    plt.grid(True, linestyle=":", alpha=0.6)
    plt.tight_layout()
    plt.show()

    print(f"[{file.name}] LE={x_le:.4f}, TE={x_te:.4f}, chord={c:.4f}")
